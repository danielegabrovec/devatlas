using System;
using System.IO;
using System.Linq;
using System.Text.Json;
using System.Threading.Tasks;
using System.Collections.ObjectModel;
using Avalonia;
using Avalonia.Controls;
using Avalonia.Controls.ApplicationLifetimes;
using Avalonia.Layout;
using Avalonia.Media;
using Avalonia.Themes.Fluent;
using Avalonia.Threading;
namespace RecipeApp;
class Program { [STAThread] public static void Main(string[] args)=>AppBuilder.Configure<App>().UsePlatformDetect().StartWithClassicDesktopLifetime(args); }
class App:Application {
 public override void Initialize(){Styles.Add(new FluentTheme());}
 public override void OnFrameworkInitializationCompleted(){if(ApplicationLifetime is IClassicDesktopStyleApplicationLifetime desktop) desktop.MainWindow=new MainWindow(); base.OnFrameworkInitializationCompleted();}
}
class MainWindow:Window {
 public MainWindow(){Title="Ricetta Avalonia";Width=760;Height=520; var panel=new StackPanel{Spacing=12,Margin=new Thickness(20)};
var b=new Button{Content="Salva preferenze"}; var label=new TextBlock(); b.Click+=async (_,_)=>{try{var dir=Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),"AvaloniaRecipe"); Directory.CreateDirectory(dir); var path=Path.Combine(dir,"settings.json"); await File.WriteAllTextAsync(path,JsonSerializer.Serialize(new { theme="dark",zoom=110})); label.Text=path;}catch(Exception ex){label.Text=ex.Message;}}; panel.Children.Add(b); panel.Children.Add(label);
 Content=new ScrollViewer{Content=panel}; }
}
