using System;
using System.Linq;
using System.Threading.Tasks;
using System.Collections.ObjectModel;
using Microsoft.UI.Xaml;
using Microsoft.UI.Xaml.Controls;
namespace RecipeApp;
public class MainWindow:Window {
 public MainWindow(){Title="Ricetta WinUI"; var panel=new StackPanel{Spacing=12,Margin=new Thickness(20)};
var all=new[]{"Python","PySide6","PostgreSQL","Rust"}; var search=new AutoSuggestBox{PlaceholderText="Cerca"}; search.TextChanged+=(sender,e)=>{if(e.Reason==AutoSuggestionBoxTextChangeReason.UserInput) sender.ItemsSource=all.Where(x=>x.Contains(sender.Text,StringComparison.OrdinalIgnoreCase)).ToArray();}; panel.Children.Add(search);
 Content=panel; }
}
