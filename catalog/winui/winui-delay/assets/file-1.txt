using System;
using System.Linq;
using System.Threading.Tasks;
using System.Collections.ObjectModel;
using Microsoft.UI.Xaml;
using Microsoft.UI.Xaml.Controls;
namespace RecipeApp;
public class MainWindow:Window {
 public MainWindow(){Title="Ricetta WinUI"; var panel=new StackPanel{Spacing=12,Margin=new Thickness(20)};
var b=new Button{Content="Attendi"}; b.Click+=async (_,_)=>{b.IsEnabled=false; try{await Task.Delay(1000); b.Content="Pronto";}finally{b.IsEnabled=true;}}; panel.Children.Add(b);
 Content=panel; }
}
