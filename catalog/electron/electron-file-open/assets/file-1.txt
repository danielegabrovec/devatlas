const {app,BrowserWindow,ipcMain,dialog,clipboard,nativeTheme,Menu,screen,shell,Notification,powerMonitor}=require('electron');
const fs=require('node:fs/promises');
const path=require('node:path');
const {pathToFileURL}=require('node:url');
let win;
app.whenReady().then(async()=>{
 win=new BrowserWindow({width:900,height:620,webPreferences:{preload:path.join(__dirname,'preload.cjs'),contextIsolation:true,nodeIntegration:false,sandbox:true}});
 win.webContents.setWindowOpenHandler(()=>({action:'deny'}));
 win.webContents.on('will-navigate',event=>event.preventDefault());
 ipcMain.handle('demo:run',async(event)=>{
  if(event.senderFrame?.url!==pathToFileURL(path.join(__dirname,'index.html')).href) throw new Error('Origine IPC non consentita');
  const choice=await dialog.showOpenDialog(win,{properties:["openFile"],filters:[{name:"Testo",extensions:["txt","md"]}]}); if(choice.canceled) return "Annullato"; const stat=await fs.stat(choice.filePaths[0]); if(stat.size>1000000) throw new Error("File oltre 1 MB"); return await fs.readFile(choice.filePaths[0],"utf8");
 });
 await win.loadFile('index.html');
});
app.on('window-all-closed',()=>{if(process.platform!=='darwin') app.quit();});
app.on('activate',()=>{if(BrowserWindow.getAllWindows().length===0) app.relaunch(),app.exit(0);});
