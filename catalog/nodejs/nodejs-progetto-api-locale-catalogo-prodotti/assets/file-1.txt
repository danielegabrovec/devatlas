import{createServer}from'node:http';import{products,find}from'./products.mjs';const s=createServer((q,r)=>{const u=new URL(q.url,'http://localhost');const id=/^\/products\/(\d+)$/.exec(u.pathname);const data=u.pathname==='/products'?products:id?find(Number(id[1])):null;r.writeHead(data?200:404,{'content-type':'application/json'});r.end(JSON.stringify(data??{error:'not found'}))});s.listen(3000,'127.0.0.1');process.once('SIGINT',()=>s.close());
