#[derive(Debug)]
struct DomainError(&'static str);
impl std::fmt::Display for DomainError{fn fmt(&self,f:&mut std::fmt::Formatter<'_>)->std::fmt::Result{write!(f,"{}",self.0)}}
impl std::error::Error for DomainError{}
fn main() -> Result<(), Box<dyn std::error::Error>> {
let error=DomainError("quantita negativa");println!("{}",error);
Ok(())
}
