import test from 'node:test';import assert from 'node:assert/strict';import {addTask,completed} from '../src/model.mjs';
test('rifiuta un titolo vuoto',()=>assert.throws(()=>addTask([],'  ')));
test('assegna ID e non muta origine',()=>{const old=[{id:4,title:'prima',done:true}];const next=addTask(old,' seconda ');assert.equal(next[1].id,5);assert.equal(next[1].title,'seconda');assert.equal(old.length,1);});
test('conta soltanto completate',()=>assert.equal(completed([{done:true},{done:false}]),1));
