#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *split=new QSplitter; split->addWidget(new QPlainTextEdit("Sorgente")); split->addWidget(new QPlainTextEdit("Risultato")); layout->addWidget(split);
 window.show(); return app.exec();
}
