#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *b=new QPushButton("Colore"); layout->addWidget(b); QObject::connect(b,&QPushButton::clicked,&window,[&window,b]{auto c=QColorDialog::getColor(Qt::blue,&window); if(c.isValid()) b->setStyleSheet("background:"+c.name());});
 window.show(); return app.exec();
}
