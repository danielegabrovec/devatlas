#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *edit=new QLineEdit; edit->setPlaceholderText("Ricerca"); auto *shortcut=new QShortcut(QKeySequence::Find,&window); QObject::connect(shortcut,&QShortcut::activated,edit,[edit]{edit->setFocus();}); layout->addWidget(edit);
 window.show(); return app.exec();
}
