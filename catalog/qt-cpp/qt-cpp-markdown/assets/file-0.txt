#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *view=new QTextBrowser; view->setMarkdown("# Guida\n\n- Compila\n- Avvia"); layout->addWidget(view);
 window.show(); return app.exec();
}
