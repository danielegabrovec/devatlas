#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *edit=new QLineEdit; edit->setValidator(new QIntValidator(1,120,edit)); edit->setPlaceholderText("1–120"); layout->addWidget(edit);
 window.show(); return app.exec();
}
