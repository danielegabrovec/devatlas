#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *combo=new QComboBox; combo->addItems({"localhost","server"}); combo->setEditable(true); combo->setInsertPolicy(QComboBox::NoInsert); layout->addWidget(combo);
 window.show(); return app.exec();
}
