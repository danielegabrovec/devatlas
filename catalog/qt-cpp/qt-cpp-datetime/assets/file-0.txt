#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *edit=new QDateTimeEdit(QDateTime::currentDateTime()); edit->setDisplayFormat("dd/MM/yyyy HH:mm"); edit->setCalendarPopup(true); layout->addWidget(edit);
 window.show(); return app.exec();
}
