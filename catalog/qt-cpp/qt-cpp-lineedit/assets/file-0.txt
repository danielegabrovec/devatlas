#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *edit=new QLineEdit; auto *label=new QLabel; layout->addWidget(edit); layout->addWidget(label); QObject::connect(edit,&QLineEdit::textChanged,label,&QLabel::setText);
 window.show(); return app.exec();
}
