#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *b=new QPushButton("Nome"); auto *label=new QLabel; layout->addWidget(b); layout->addWidget(label); QObject::connect(b,&QPushButton::clicked,&window,[&window,label]{bool ok=false; auto text=QInputDialog::getText(&window,"Nome","Nome:",QLineEdit::Normal,{},&ok); if(ok) label->setText(text);});
 window.show(); return app.exec();
}
