#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *edit=new QLineEdit; edit->setPlaceholderText("Password"); edit->setEchoMode(QLineEdit::Password); layout->addWidget(edit);
 window.show(); return app.exec();
}
