#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *model=new QStringListModel(QStringList{"Uno","Due"},&window); auto *view=new QListView; view->setModel(model); auto *label=new QLabel; QObject::connect(view->selectionModel(),&QItemSelectionModel::currentChanged,label,[label](const QModelIndex &now,const QModelIndex &){label->setText(now.data().toString());}); layout->addWidget(view); layout->addWidget(label);
 window.show(); return app.exec();
}
