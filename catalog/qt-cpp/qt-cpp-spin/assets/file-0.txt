#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *spin=new QSpinBox; spin->setRange(0,200); spin->setSingleStep(5); spin->setSuffix(" pezzi"); layout->addWidget(spin);
 window.show(); return app.exec();
}
