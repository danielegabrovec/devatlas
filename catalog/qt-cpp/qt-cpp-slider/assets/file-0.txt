#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *slider=new QSlider(Qt::Horizontal); auto *bar=new QProgressBar; slider->setRange(0,100); QObject::connect(slider,&QSlider::valueChanged,bar,&QProgressBar::setValue); layout->addWidget(slider); layout->addWidget(bar);
 window.show(); return app.exec();
}
