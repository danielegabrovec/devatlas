#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *table=new QTableWidget(2,2); table->setHorizontalHeaderLabels({"Nome","Valore"}); table->setItem(0,0,new QTableWidgetItem("Uno")); table->setItem(0,1,new QTableWidgetItem("1")); layout->addWidget(table);
 window.show(); return app.exec();
}
