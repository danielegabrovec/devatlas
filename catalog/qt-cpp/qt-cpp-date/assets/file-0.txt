#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *edit=new QDateEdit(QDate::currentDate()); edit->setCalendarPopup(true); edit->setDisplayFormat("dd/MM/yyyy"); layout->addWidget(edit);
 window.show(); return app.exec();
}
