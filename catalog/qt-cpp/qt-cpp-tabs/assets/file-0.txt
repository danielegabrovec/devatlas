#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *tabs=new QTabWidget; tabs->addTab(new QPlainTextEdit,"Codice"); tabs->addTab(new QLabel("Opzioni"),"Impostazioni"); layout->addWidget(tabs);
 window.show(); return app.exec();
}
