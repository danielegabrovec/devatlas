#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *form=new QFormLayout; form->addRow("Nome",new QLineEdit); form->addRow("Email",new QLineEdit); layout->addLayout(form);
 window.show(); return app.exec();
}
