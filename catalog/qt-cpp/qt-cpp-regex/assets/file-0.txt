#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *edit=new QLineEdit; edit->setValidator(new QRegularExpressionValidator(QRegularExpression("[A-Z]{2}[0-9]{4}"),edit)); edit->setPlaceholderText("AB1234"); layout->addWidget(edit);
 window.show(); return app.exec();
}
