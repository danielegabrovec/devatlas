#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *settings=new QSettings("Ricettario","CppDemo",&window); auto *check=new QCheckBox("Dettagli"); check->setChecked(settings->value("details",false).toBool()); QObject::connect(check,&QCheckBox::toggled,settings,[settings](bool v){settings->setValue("details",v);}); layout->addWidget(check);
 window.show(); return app.exec();
}
