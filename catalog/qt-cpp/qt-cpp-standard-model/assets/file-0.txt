#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *model=new QStandardItemModel(&window); model->setHorizontalHeaderLabels({"Nome","Ruolo"}); model->appendRow({new QStandardItem("Ada"),new QStandardItem("Sviluppo")}); auto *view=new QTableView; view->setModel(model); layout->addWidget(view);
 window.show(); return app.exec();
}
