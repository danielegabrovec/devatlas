#include <QtWidgets>
#include <QtCore>
#include <QtGui>

int main(int argc, char **argv) {
 QApplication app(argc,argv);
 QWidget window; window.resize(720,480);
 auto *layout=new QVBoxLayout(&window);
 auto *spin=new QDoubleSpinBox; spin->setRange(0,9999); spin->setDecimals(2); spin->setPrefix("EUR "); layout->addWidget(spin);
 window.show(); return app.exec();
}
