export function parseTSV(text){const lines=text.trim().split(/\r?\n/);if(lines.shift()!=='nome\timporto')throw Error('Intestazione richiesta: nome TAB importo');return lines.filter(Boolean).map((line,i)=>{const[name,raw,...extra]=line.split('\t');const amount=Number(raw);if(!name||raw===undefined||!Number.isFinite(amount)||extra.length)throw Error('Riga invalida '+(i+2));return{name,amount}})}
