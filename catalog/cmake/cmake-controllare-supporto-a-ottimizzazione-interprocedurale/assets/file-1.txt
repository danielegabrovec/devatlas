cmake_minimum_required(VERSION 3.24)
project(RecipeBuild VERSION 1.2.0 LANGUAGES CXX)
add_executable(recipe main.cpp)
include(CheckIPOSupported)
check_ipo_supported(RESULT recipe_ipo OUTPUT recipe_ipo_error)
if(recipe_ipo)
 set_property(TARGET recipe PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
else()
 message(STATUS "IPO unavailable: ${recipe_ipo_error}")
endif()
