import {createClient} from 'redis';import {localUrl} from './model.mjs';export async function connect(){const uri=localUrl(process.env.ATLAS_REDIS_URL||'redis://127.0.0.1:6381','redis:',6381);const client=createClient({url:uri,socket:{connectTimeout:2000,reconnectStrategy:false}});client.on('error',error=>console.error(error.message));await client.connect();return{client,items:client};}
