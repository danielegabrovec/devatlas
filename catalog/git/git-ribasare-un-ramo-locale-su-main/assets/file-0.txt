$work=Join-Path ([IO.Path]::GetTempPath()) ("git-recipe-"+[guid]::NewGuid())
git init -b main $work
Set-Location -LiteralPath $work
git config user.name "Autore Demo"
git config user.email "demo@example.invalid"
Set-Content README.md "# Demo"
git add README.md
git commit -m "Iniziale"
git switch -c feature/note
Set-Content note.txt "Nota"
git add note.txt
git commit -m "Nota feature"
git switch main
Set-Content config.txt "config"
git add config.txt
git commit -m "Config principale"
git switch feature/note
git rebase main
git log --oneline --graph
