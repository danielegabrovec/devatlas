CREATE TABLE people(id INTEGER PRIMARY KEY, name VARCHAR(80) NOT NULL, city VARCHAR(80), age INTEGER, email VARCHAR(120) UNIQUE) ENGINE=InnoDB;
CREATE TABLE products(id INTEGER PRIMARY KEY,name VARCHAR(80),price DECIMAL(10,2),stock INTEGER) ENGINE=InnoDB;
CREATE TABLE orders(id INTEGER PRIMARY KEY,person_id INTEGER,total DECIMAL(10,2),status VARCHAR(20),created_at VARCHAR(30),FOREIGN KEY(person_id) REFERENCES people(id)) ENGINE=InnoDB;
CREATE TABLE order_items(order_id INTEGER,product_id INTEGER,qty INTEGER,PRIMARY KEY(order_id,product_id),FOREIGN KEY(order_id) REFERENCES orders(id),FOREIGN KEY(product_id) REFERENCES products(id)) ENGINE=InnoDB;
