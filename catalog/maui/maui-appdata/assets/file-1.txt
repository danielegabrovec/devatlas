using System;
using System.IO;
using System.Linq;
using System.Threading.Tasks;
using System.Collections.ObjectModel;
using Microsoft.Maui.Controls;
using Microsoft.Maui.Graphics;
using Microsoft.Maui.ApplicationModel.DataTransfer;
using Microsoft.Maui.Storage;
namespace RecipeApp;
public class MainPage:ContentPage {
 public MainPage(){ Title="Ricetta MAUI"; var panel=new VerticalStackLayout{Spacing=12,Padding=20};
var label=new Label(); var b=new Button{Text="Salva appunto"}; b.Clicked+=async (_,_)=>{try{var path=Path.Combine(FileSystem.AppDataDirectory,"note.txt"); await File.WriteAllTextAsync(path,"Nota UTF-8"); label.Text=path;}catch(Exception ex){label.Text=ex.Message;}}; panel.Children.Add(b); panel.Children.Add(label);
 Content=new ScrollView{Content=panel}; }
}
