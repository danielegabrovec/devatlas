using System;
using System.IO;
using System.Linq;
using System.Threading.Tasks;
using System.Collections.ObjectModel;
using Microsoft.Maui.Controls;
using Microsoft.Maui.Graphics;
using Microsoft.Maui.ApplicationModel.DataTransfer;
using Microsoft.Maui.Storage;
namespace RecipeApp;
public class MainPage:ContentPage {
 public MainPage(){ Title="Ricetta MAUI"; var panel=new VerticalStackLayout{Spacing=12,Padding=20};
var b=new Button{Text="Attendi"}; b.Clicked+=async (_,_)=>{b.IsEnabled=false; try{await Task.Delay(1000); b.Text="Pronto";}finally{b.IsEnabled=true;}}; panel.Children.Add(b);
 Content=new ScrollView{Content=panel}; }
}
