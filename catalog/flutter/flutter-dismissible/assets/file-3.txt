import 'package:flutter/material.dart';

void main()=>runApp(const AtlasApp());
class AtlasApp extends StatefulWidget{const AtlasApp({super.key});@override State<AtlasApp> createState()=>_AtlasState();}
class _AtlasState extends State<AtlasApp>{
ThemeMode appTheme=ThemeMode.light;
final rows=<int>[1,2,3];
@override Widget build(BuildContext context)=>MaterialApp(theme:ThemeData(colorScheme:ColorScheme.fromSeed(seedColor:Colors.indigo)),darkTheme:ThemeData.dark(),themeMode:appTheme,home:Builder(builder:(context)=>Scaffold(key:const Key('page'),appBar:AppBar(title:const Text('Laboratorio Flutter')),body:Padding(padding:const EdgeInsets.all(20),child:ListView(children:rows.map((id)=>Dismissible(key:ValueKey(id),background:Container(color:Colors.red),onDismissed:(_)=>setState(()=>rows.remove(id)),child:ListTile(title:Text('Riga $id')))).toList())))));
}
