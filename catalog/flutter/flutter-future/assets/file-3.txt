import 'package:flutter/material.dart';

void main()=>runApp(const AtlasApp());
class AtlasApp extends StatefulWidget{const AtlasApp({super.key});@override State<AtlasApp> createState()=>_AtlasState();}
class _AtlasState extends State<AtlasApp>{
ThemeMode appTheme=ThemeMode.light;
final Future<String> data=Future.delayed(const Duration(milliseconds:300),()=> 'Dati locali');
@override Widget build(BuildContext context)=>MaterialApp(theme:ThemeData(colorScheme:ColorScheme.fromSeed(seedColor:Colors.indigo)),darkTheme:ThemeData.dark(),themeMode:appTheme,home:Builder(builder:(context)=>Scaffold(key:const Key('page'),appBar:AppBar(title:const Text('Laboratorio Flutter')),body:Padding(padding:const EdgeInsets.all(20),child:FutureBuilder<String>(future:data,builder:(context,snapshot){if(snapshot.hasError)return Text('Errore: ${snapshot.error}');if(!snapshot.hasData)return const CircularProgressIndicator();return Text(snapshot.data!);})))));
}
