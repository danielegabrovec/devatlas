import 'package:flutter/material.dart';

ThemeMode appTheme=ThemeMode.light;
double value=30;
@override Widget build(BuildContext context)=>MaterialApp(theme:ThemeData(colorScheme:ColorScheme.fromSeed(seedColor:Colors.indigo)),darkTheme:ThemeData.dark(),themeMode:appTheme,home:Builder(builder:(context)=>Scaffold(key:const Key('page'),appBar:AppBar(title:const Text('Laboratorio Flutter')),body:Padding(padding:const EdgeInsets.all(20),child:Column(children:[Slider(value:value,min:0,max:100,divisions:20,onChanged:(next)=>setState(()=>value=next)),Text('${value.toStringAsFixed(0)}%')])))));
