import 'package:flutter/material.dart';

void main()=>runApp(const AtlasApp());
class AtlasApp extends StatefulWidget{const AtlasApp({super.key});@override State<AtlasApp> createState()=>_AtlasState();}
class _AtlasState extends State<AtlasApp>{
ThemeMode appTheme=ThemeMode.light;

@override Widget build(BuildContext context)=>MaterialApp(theme:ThemeData(colorScheme:ColorScheme.fromSeed(seedColor:Colors.indigo)),darkTheme:ThemeData.dark(),themeMode:appTheme,home:Builder(builder:(context)=>Scaffold(key:const Key('page'),appBar:AppBar(title:const Text('Laboratorio Flutter')),body:Padding(padding:const EdgeInsets.all(20),child:SwitchListTile(title:const Text('Tema scuro'),value:appTheme==ThemeMode.dark,onChanged:(value)=>setState(()=>appTheme=value?ThemeMode.dark:ThemeMode.light))))));
}
