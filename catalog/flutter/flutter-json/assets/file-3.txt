import 'package:flutter/material.dart';
import 'dart:convert';
void main()=>runApp(const AtlasApp());
class AtlasApp extends StatefulWidget{const AtlasApp({super.key});@override State<AtlasApp> createState()=>_AtlasState();}
class _AtlasState extends State<AtlasApp>{
ThemeMode appTheme=ThemeMode.light;
String parse(){final value=jsonDecode('{"name":"Ada","age":35}');if(value is! Map<String,dynamic>||value['name'] is! String||value['age'] is! int)throw const FormatException('Dati non validi');return '${value['name']} · ${value['age']}';}
@override Widget build(BuildContext context)=>MaterialApp(theme:ThemeData(colorScheme:ColorScheme.fromSeed(seedColor:Colors.indigo)),darkTheme:ThemeData.dark(),themeMode:appTheme,home:Builder(builder:(context)=>Scaffold(key:const Key('page'),appBar:AppBar(title:const Text('Laboratorio Flutter')),body:Padding(padding:const EdgeInsets.all(20),child:Text(parse())))));
}
