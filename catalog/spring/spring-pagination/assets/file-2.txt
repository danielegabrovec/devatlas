package it.atlas;
import java.util.*;
import java.util.concurrent.*;
import java.net.URI;
import org.springframework.web.bind.annotation.*;
import org.springframework.http.*;
import org.springframework.web.server.ResponseStatusException;
import org.springframework.beans.factory.annotation.*;
import org.springframework.stereotype.*;
import jakarta.validation.Valid;
import jakarta.validation.constraints.*;
@RestController
public class DemoController {
private final List<String> rows=List.of("Ada","Grace","Linus"); @GetMapping("/people") public List<String> list(@RequestParam(name="offset",defaultValue="0") int offset,@RequestParam(name="limit",defaultValue="2") int limit){if(offset<0||limit<1||limit>100)throw new ResponseStatusException(HttpStatus.BAD_REQUEST,"Limiti non validi");return rows.stream().skip(offset).limit(limit).toList();}
}
