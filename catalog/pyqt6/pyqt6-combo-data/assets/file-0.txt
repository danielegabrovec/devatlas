import sys, json, sqlite3, tempfile, logging
from pathlib import Path
from PyQt6.QtCore import *
from PyQt6.QtGui import *
from PyQt6.QtWidgets import *
Signal = pyqtSignal
Slot = pyqtSlot

def build(root, layout):
    combo = QComboBox()
    combo.addItem("Produzione", "prod")
    combo.addItem("Sviluppo", "dev")
    label = QLabel("prod")
    combo.currentIndexChanged.connect(lambda _: label.setText(combo.currentData()))
    layout.addWidget(combo)
    layout.addWidget(label)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = QWidget()
    window.setWindowTitle("Ricetta Qt")
    window.resize(720, 480)
    layout = QVBoxLayout(window)
    build(window, layout)
    window.show()
    sys.exit(app.exec())
