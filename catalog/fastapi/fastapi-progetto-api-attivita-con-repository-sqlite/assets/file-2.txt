from fastapi.testclient import TestClient
from app import app
import repository
def test_flow(tmp_path,monkeypatch):
    monkeypatch.setattr(repository,'DB',tmp_path/'test.sqlite')
    with TestClient(app) as c:
        row=c.post('/tasks',json={'title':'Prova'}).json()
        assert c.patch('/tasks/'+str(row['id'])).status_code==200
        assert c.get('/tasks').json()[0]['done']==1
